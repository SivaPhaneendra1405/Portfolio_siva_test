@Test
public void testResumeCreationFlow() {
    // Step 1: Open Bowizzy & Login
    driver.get("https://bowizzy.com/login");
    driver.findElement(By.id("email")).sendKeys("testuser@gmail.com");
    driver.findElement(By.name("password")).sendKeys("Pass123!");
    driver.findElement(By.cssSelector("button[type='submit']")).click();
    
    // Step 2: Navigate to Resume Builder
    wait.until(ExpectedConditions.urlContains("/dashboard"));
    driver.findElement(By.xpath("//a[contains(text(),'Create Resume')]")).click();
    
    // Step 3: Select ATS Template & Verify UI
    driver.findElement(By.className("template-card-ats")).click();
    Assert.assertTrue(driver.findElement(By.id("resume-editor")).isDisplayed());
    
    // Step 4: Fill Personal Info & Trigger Save
    driver.findElement(By.id("fullName")).sendKeys("Siva Phaneendra");
    driver.findElement(By.id("save-resume")).click();
    
    // Step 5: Assert Success Toast Alert
    WebElement toast = wait.until(ExpectedConditions.visibilityOfElementLocated(
        By.xpath("//*[contains(text(),'Saved Successfully')]")));
    Assert.assertNotNull(toast);
}
